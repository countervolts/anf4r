cmake_minimum_required(VERSION 3.20)
project(anf4r CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  # anf4r.dll for OptiScaler under Proton: Vulkan entry points come from the game's device
  add_library(anf4r SHARED runtime/vk.cpp runtime/vkfn.cpp runtime/pass.cpp runtime/fgnet.cpp runtime/fg.cpp runtime/sr.cpp runtime/anf4r_api.cpp)
  target_compile_definitions(anf4r PRIVATE ANF4R_DYNAMIC_VK VK_NO_PROTOTYPES)
  target_include_directories(anf4r PRIVATE runtime ${ANF4R_VULKAN_INCLUDE})
  target_link_options(anf4r PRIVATE -static -static-libgcc -static-libstdc++)
  set_target_properties(anf4r PROPERTIES PREFIX "")
  return()
endif()
find_package(Vulkan REQUIRED)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(anf4r_rt STATIC runtime/vk.cpp runtime/pass.cpp)
target_link_libraries(anf4r_rt PUBLIC Vulkan::Vulkan)
add_executable(smoke_pass tests/smoke_pass.cpp)
target_link_libraries(smoke_pass PRIVATE anf4r_rt)
target_sources(anf4r_rt PRIVATE runtime/fgnet.cpp)
target_sources(anf4r_rt PRIVATE runtime/fg.cpp)
add_executable(fg_test tests/fg_test.cpp)
target_link_libraries(fg_test PRIVATE anf4r_rt)
target_sources(anf4r_rt PRIVATE runtime/sr.cpp)
add_executable(sr_run tools/sr_run.cpp)
target_link_libraries(sr_run PRIVATE anf4r_rt)
add_executable(fgnet_test tests/fgnet_test.cpp)
target_link_libraries(fgnet_test PRIVATE anf4r_rt)

# Compile our own compute kernels to SPIR-V (the frame-generation shaders are used as-is from data/spirv).
find_program(GLSLC glslc REQUIRED)
set(ANF_KERNELS conv_wmma sr_stab fg_conv_i8 fg_head_lr fg_head_qr fg_resample sr_pre sr_post sr_resize)
set(ANF_SPV "")
foreach(k ${ANF_KERNELS})
  add_custom_command(OUTPUT ${CMAKE_SOURCE_DIR}/build/${k}.spv
    COMMAND ${GLSLC} --target-env=vulkan1.3 -O ${CMAKE_SOURCE_DIR}/kernels/${k}.comp -o ${CMAKE_SOURCE_DIR}/build/${k}.spv
    DEPENDS ${CMAKE_SOURCE_DIR}/kernels/${k}.comp)
  list(APPEND ANF_SPV ${CMAKE_SOURCE_DIR}/build/${k}.spv)
endforeach()
add_custom_target(anf_kernels ALL DEPENDS ${ANF_SPV})

# C API library (Linux .so for tests; the Windows DLL for OptiScaler is built with cmake/mingw-w64.cmake)
add_library(anf4r SHARED runtime/anf4r_api.cpp)
target_link_libraries(anf4r PRIVATE anf4r_rt)
add_executable(profile tools/profile.cpp)
target_link_libraries(profile PRIVATE anf4r_rt)
