cmake_minimum_required(VERSION 3.20)
project(anf4r CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
add_library(anf4r_rt STATIC runtime/vk.cpp runtime/pass.cpp)
target_link_libraries(anf4r_rt PUBLIC Vulkan::Vulkan)
add_executable(smoke_pass tests/smoke_pass.cpp)
target_link_libraries(smoke_pass PRIVATE anf4r_rt)
add_executable(conv_test tests/conv_test.cpp)
target_link_libraries(conv_test PRIVATE anf4r_rt)
target_sources(anf4r_rt PRIVATE runtime/model.cpp)
add_executable(model_test tests/model_test.cpp)
target_link_libraries(model_test PRIVATE anf4r_rt)
